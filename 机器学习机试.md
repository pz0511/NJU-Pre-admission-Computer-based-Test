```python
def k_means_clustering(points, k, initial_centroids, max_iterations):
    points = np.array(points)
    centroids = np.array(initial_centroids)
    for iteration in range(max_iterations):
        # Assign points to the nearest centroid
        distances = np.array([euclidean_distance(points, centroid) for centroid in centroids])
        assignments = np.argmin(distances, axis=0)#把每个点分到最近的质心
        new_centroids = np.array([points[assignments == i].mean(axis=0) if len(points[assignments == i]) > 0 else centroids[i] for i in range(k)])
        # Check for convergence
        if np.all(centroids == new_centroids):
            break
        centroids = new_centroids
        centroids = np.round(centroids,4)
    return [tuple(centroid) for centroid in centroids]
```

```python
    #得到所有指数组合
```

### ML31 梯度下降

```py
def gradient_descent(X, y, weights, learning_rate, n_iterations, batch_size=1, method='batch'):
    m=X.shape[0]
    for _ in range(n_iterations):
        #批量梯度下降
        if method == 'batch':
            grad=(2/m)*X.T@(X@weights-y)
            weights-=learning_rate*grad
        #随机梯度下降
        elif method == 'stochastic':
            for i in range(m):                     # 遍历所有样本
                xi = X[i:i+1]#保持维度
                yi = y[i:i+1]
                grad = 2 * xi.T @ (xi @ weights - yi)
                weights -= learning_rate * grad
        #小批量梯度下降
        elif method == 'mini_batch':
            for i in range(0, m, batch_size):      # 顺序划分批次（不随机）
                xi = X[i:i+batch_size]
                yi = y[i:i+batch_size]
                grad = (2 / len(xi)) * xi.T @ (xi @ weights - yi)
                weights -= learning_rate * grad
    return weights
```

### ML32 梯度下降实现Lasso回归(L_1正则化线性回归)

```python
def l1_regularization_gradient_descent(X: np.array, y: np.array, alpha: float = 0.1, learning_rate: float = 0.01, max_iter: int = 1000, tol: float = 1e-4) -> tuple:
    m,n=X.shape
    weights = np.zeros(n)   # 初始权重为 0
    bias = 0.0
    for _ in range(max_iter):
        y_pred=X@weights+bias
        err=y_pred-y
        gred_w=(1/m)*X.T@err#经验风险w的梯度
        grad_b=(1/m)*np.sum(err)#经验风险b的梯度
        grad_reg=alpha*np.sign(weights)#结构风险w的梯度
        gred_w=gred_w+grad_reg
        new_w=weights-gred_w*learning_rate
        new_b=bias-grad_b*learning_rate
        # if (np.linalg.norm(new_w-weights)<tol and abs(new_b-bias)<tol):
        #     weights, bias = new_w, new_b
        #     break
        weights, bias = new_w, new_b
    return [round(w, 3) for w in weights], round(bias, 3)
```

```python
import numpy as np
from collections import Counter
def majority_label(examples,target_attr):#出现最多的标签
    labels=[ex[target_attr] for ex in examples]
    return Counter(labels).most_common(1)[0][0]#most_common得到的元素是(标签,次数)
def entropy(labels):#信息熵
    v,c=np.unique(np.array(labels),return_counts=True)
    prob=c/len(labels)
    return -np.sum(prob@np.log2(prob))
def info_gain(examples,attr,target_attr):#信息增益
    #总信息熵
    labels=[ex[target_attr] for ex in examples]
    HS=entropy(labels)
    #attr的信息熵
    hs_attr=0
    feature_col=np.array([ex[attr] for ex in examples])
    values,counts=np.unique(feature_col,return_counts=True)
    for v,c in zip(values,counts):
        sub_examples = [ex for ex in examples if ex[attr] == v]
        sub_labels = [ex[target_attr] for ex in sub_examples]
        hs_attr+=(c/len(examples))*entropy(sub_labels)#注意这里是加。因为entropy返回的就是带负号
    return HS-hs_attr
def learn_decision_tree(examples: list[dict], attributes: list[str], target_attr: str) -> dict:
    label=[ex[target_attr] for ex in examples]
    #终止条件①：如果当前节点所有样本的标签完全一致，直接返回该标签（纯叶子）
    if len(set(label))==1:
        return label[0]
    #终止条件②：如果没有可用的特征继续分裂，则返回当前样本中的多数类（投票）
    if not attributes:
        return majority_label(examples,target_attr)
    #得到最佳划分特征
    best_attr=max(attributes,key=lambda a:info_gain(examples,a,target_attr))
    #初始化当前树
    decision_tree = {best_attr: {}}
    for value in set([ex[best_attr] for ex in examples]):
        sub_examples=[ex for ex in examples if ex[best_attr]==value]#传入下一次递归
        sub_attributes=[attr for attr in attributes if attr!=best_attr]#传入下一次递归
        decision_tree[best_attr][value]=learn_decision_tree(sub_examples,sub_attributes,target_attr)
    return decision_tree

```

### DL7 两个正态分布的KL散度

**KL散度 $D_{KL}(P \parallel Q)$** 衡量的是：**因为误用分布 Q 来编码来自 P 的样本，导致需要额外消耗多少信息量（或损失了多少信息）。**

 是衡量**两个概率分布之间差异**的非对称度量。
$$
D_{KL}(P \parallel Q) = \sum_{i} P(i) \cdot \log \frac{P(i)}{Q(i)}
$$

```python
def kl_divergence(p, q):
    # 避免除以0或log(0)导致nan
    p = np.array(p, dtype=np.float64)
    q = np.array(q, dtype=np.float64)
    return np.sum(p * np.log(p / q + 1e-10))  # 加极小值防除零
#正态分布时
def kl_divergence_normal(mu_p, sigma_p, mu_q, sigma_q):
    kl_div=((mu_p-mu_q)**2)/(sigma_q**2)+sigma_p**2/sigma_q**2-1+np.log(sigma_q**2/sigma_p**2)
    return kl_div/2
```

```python
def simple_conv2d(input_matrix: np.ndarray, kernel: np.ndarray, padding: int, stride: int):
    # 1. 获取各维度大小
    input_h, input_w = input_matrix.shape
    kernel_h, kernel_w = kernel.shape
    # 2. 计算输出尺寸
    output_h = (input_h + 2 * padding - kernel_h) // stride + 1
    output_w = (input_w + 2 * padding - kernel_w) // stride + 1
    # 3. Padding：四周填充0
    padded = np.pad(input_matrix, 
                    pad_width=((padding, padding), (padding, padding)), 
                    mode='constant')
    # 4. 创建输出矩阵
    output_matrix = np.zeros((output_h, output_w))
    # 5. 滑动窗口计算
    for i in range(output_h):
        for j in range(output_w):
            # 窗口在padded矩阵中的起始位置
            row_start = i * stride
            col_start = j * stride
            # 取出当前窗口
            window = padded[row_start : row_start + kernel_h,
                            col_start : col_start + kernel_w]
            # 对应元素相乘再求和
            output_matrix[i, j] = np.sum(window * kernel)
    return output_matrix
```

### DL14 Adam优化算法

```python
def adam_optimizer(f, grad, x0, learning_rate=0.001, beta1=0.9, beta2=0.999,
                   epsilon=1e-8, num_iterations=10):
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    
    for t in range(1, num_iterations + 1):
        g = grad(x)                     # 当前梯度
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g ** 2)
        
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        x = x - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
    return x
```

1.  **计算当前梯度**：$ g_t = \nabla_\theta f_t(\theta_{t-1}) $
2.  **更新有偏一阶矩估计（动量）**：$ m_t = \beta_1 \cdot m_{t-1} + (1 - \beta_1) \cdot g_t $
3.  **更新有偏二阶矩估计（自适应率）**：$ v_t = \beta_2 \cdot v_{t-1} + (1 - \beta_2) \cdot g_t^2 $（注意这里是逐元素平方）
4.  **偏差校正（Bias Correction）**：由于 $ m_0 $ 和 $ v_0 $ 初始化为 0，在训练初期它们会偏向 0，所以需要放缩。
    - $ \hat{m}_t = m_t / (1 - \beta_1^t) $
    - $ \hat{v}_t = v_t / (1 - \beta_2^t) $
5.  **更新参数**：$ \theta_t = \theta_{t-1} - \eta \cdot \frac{\hat{m}_t}{\sqrt{\hat{v}_t} + \epsilon} $

```python
	def forward(self, x, initial_hidden_state, initial_cell_state):
		"""
		x: np.ndarray, shape (T, input_size)
		initial_hidden_state: np.ndarray, shape (hidden_size, 1)
		initial_cell_state: np.ndarray, shape (hidden_size, 1)
		Returns:
			outputs: list of hidden states at each time step, each shape (hidden_size, 1)
			final_h: final hidden state
			final_c: final cell state
		"""
		h = initial_hidden_state
		c = initial_cell_state
		outputs = []
		for t in range(x.shape[0]):
			# 取出当前输入并确保列向量
			x_t = x[t].reshape(-1, 1)  # (input_size, 1)
			# 拼接 [x_t; h]
			combined = np.vstack([h, x_t])  # (input_size + hidden_size, 1)
			# 计算四个门控信号
			f = self._sigmoid(self.Wf @ combined + self.bf)
			i = self._sigmoid(self.Wi @ combined + self.bi)
			c_tilde = np.tanh(self.Wc @ combined + self.bc)
			o = self._sigmoid(self.Wo @ combined + self.bo)
			# 更新细胞状态和隐藏状态
			c = f * c + i * c_tilde
			h = o * np.tanh(c)
			# 保存当前隐藏状态（深拷贝）
			outputs.append(h.copy())
		return outputs, h, c
```

1.  **遗忘门（Forget Gate）**：决定从上一细胞状态中丢弃多少信息。
    $$
    f_t = \sigma( W_f \cdot [h_{t-1}, x_t] + b_f )
    $$
2.  **输入门（Input Gate）**：决定当前新信息中哪些需要存入细胞状态。
    $$
    i_t = \sigma( W_i \cdot [h_{t-1}, x_t] + b_i )
    $$
3.  **候选细胞状态（Candidate Cell）**：生成新的候选值向量。
    $$
    \tilde{c}_t = \tanh( W_c \cdot [h_{t-1}, x_t] + b_c )
    $$
4.  **细胞状态更新（Cell State Update）**：**这是 LSTM 最核心的加法操作，梯度在此处可以无损回流。**
    $$
    c_t = f_t \odot c_{t-1} + i_t \odot \tilde{c}_t
    $$
5.  **输出门（Output Gate）**：决定细胞状态的哪些部分输出到隐藏状态。
    
    $$
    o_t = \sigma( W_o \cdot [h_{t-1}, x_t] + b_o )
    $$
    
    $$
    h_t = o_t \odot \tanh(c_t)
    $$

```python
def adam_optimizer(parameter, grad, m, v, t, learning_rate=0.001, beta1=0.9, beta2=0.999, epsilon=1e-8):
    m = beta1 * m + (1 - beta1) * grad
```

### 19 SGDM(带动量的随机梯度下降法)

**线性模型**：$ y_{pred} = X \cdot w + b $

**损失函数**（带 $ \frac{1}{2} $ 是为了求导时消去平方项的 2，这是机器学习常见的 trick）：
$$
L = \frac{1}{2m} \sum_{i=1}^{m} (y_{pred}^{(i)} - y^{(i)})^2
$$
**梯度计算**（核心）：

- 对权重 $ w $ 求导：$ \frac{\partial L}{\partial w} = \frac{1}{m} X^T (y_{pred} - y) $
- 对偏置 $ b $ 求导：$ \frac{\partial L}{\partial b} = \frac{1}{m} \sum (y_{pred} - y) $

**带动量的参数更新（SGDM 核心）**：

> 动量 $ v $ 记录了历史梯度的“惯性”。
>
> 1. 更新速度：$ v = \text{momentum} \times v - \text{lr} \times \text{gradient} $
> 2. 更新参数：$ \text{param} = \text{param} + v $

```python
def sgdm_linear_regression(X, y, learning_rate, momentum_decay, epochs):
    m, n = X.shape
    theta = np.zeros((n, 1))
    v = np.zeros((n, 1))
    for _ in range(epochs):
        y_pred=X@theta
        grad=(1/m)*X.T@(y_pred-y)
        v=v*momentum_decay-learning_rate*grad
        theta+=v
    return np.round(theta.flatten(), 2).tolist()
```

### **DL20** **RMSProp**

线性模型：$ y_{pred} = X \cdot w + b $

损失函数（MSE，带 $ \frac{1}{2} $ 方便求导）：  
$$
L = \frac{1}{2m} \sum_{i=1}^{m} (y_{pred}^{(i)} - y^{(i)})^2
$$

梯度计算：  
$$
dw = \frac{1}{m} X^T (y_{pred} - y), \quad db = \frac{1}{m} \sum (y_{pred} - y)
$$
**RMSProp 更新公式（核心）**：

1. **缓存更新**（指数加权移动平均）：  
   $$
   v_{dw} = \beta \cdot v_{dw} + (1 - \beta) \cdot dw^2
   $$
   （$ \beta $ 是衰减率，通常取 0.9 或 0.999）

2. **参数更新**（自适应学习率）：  
   $$
   w = w - \frac{\eta}{\sqrt{v_{dw}} + \epsilon} \cdot dw
   $$
   （$ \epsilon $ 是极小常数，如 $ 10^{-8} $，防止分母为 0）

> **关键区别**：对比 SGDM（动量法）用的是**一阶矩（梯度的指数平均）**，RMSProp 用的是**二阶矩（梯度平方的指数平均）**。前者是“小球滚动惯性”，后者是“地形探测器”

```python
def rmsprop_linear_regression(X, y, learning_rate, decay_rate, epochs):
    m, n = X.shape
    theta = np.zeros((n, 1))
    s = np.zeros((n, 1))
    epsilon = 1e-8
    for _ in range(epochs):
        y_pred = X @ theta
        grad = (1 / m) * X.T @ (y_pred - y)   # (n,1)
        s = decay_rate * s + (1 - decay_rate) * grad ** 2
        theta = theta - learning_rate * grad / (np.sqrt(s) + epsilon)
    return np.round(theta.flatten(), 2).tolist()

```

### ROC曲线与AUC

```python
def calAUC(prob, labels):
    '''
    计算AUC并返回
    :param prob: 模型预测样本为Positive的概率列表，类型为ndarray
    :param labels: 样本的真实类别列表，其中1表示Positive，0表示Negtive，类型为ndarray
    :return: AUC，类型为float
    '''
    prob_labels=list(zip(prob,labels))
    prob_labels.sort(key=lambda x:x[0])
    M=np.sum(labels)
    N=len(labels)-M
    S=0#rank
    for i,x in enumerate(prob_labels):
        if x[1]==1:
            S+=(i+1)
    return (S-M*(M+1)/2)/(M*N)
```

### 感知机

**损失函数**：
$$
L(w,b)=-\sum_{x\in M}y_i(wx_i+b)
$$
感知机只针对误分类的点对参数进行更新，算法流程如下：

1. 选取初始值 $w_0, b_0$；

2. 在训练集中选取数据 $x_i, y_i$；

3. 如果 $y_i(w \cdot x_i + b) \leq 0$，即这个点为误分类点，则进行如下更新（$\eta$ 为学习率，为 $0$ 到 $1$ 之间的值）：

$$
\begin{aligned}
w &= w - \eta \frac{\partial l(w,b)}{\partial w} = w + \eta y_i x_i \\[6pt]
b &= b - \eta \frac{\partial l(w,b)}{\partial b} = b + \eta y_i
\end{aligned}
$$

4. 重复 2、3 直到训练集中没有误分类点

M为误分类点的集合

```python
class Perceptron(object):
    def __init__(self, learning_rate = 0.01, max_iter = 200):
        self.lr = learning_rate
        self.max_iter = max_iter
    def fit(self, data, label):
        self.w = np.array([1.]*data.shape[1])
        self.b = np.array([1.])
        for _ in range(self.max_iter):
            has_mis=False
            for xi,yi in zip(data,label):
                judge=yi*(self.w@xi+self.b)
                if judge<=0:#只针对误分类的点对参数进行更新
                    has_mis=True
                    self.w+=self.lr*yi*xi
                    self.b+=self.lr*yi
            if not has_mis:#停止条件
                break
        return self.w,self.b
    def predict(self, data):
        score=np.dot(data,self.w)+self.b
        predict=np.where(score>0,1,-1)#映射到1，-1
        return predict
```

```python
    X0=X[y==0]
    X1=X[y==1]

    mean0=np.mean(X0,axis=0)
```

```python
def calcInfoGain(feature, label, index):
    '''
    计算信息增益
    :param feature:测试用例中字典里的feature，类型为ndarray
    :param label:测试用例中字典里的label，类型为ndarray
    :param index:测试用例中字典里的index，即feature部分特征列的索引。该索引指的是feature中第几个特征，如index:0表示使用第一个特征来计算信息增益。
    :return:信息增益，类型float
    '''
    #先计算总的增益
    value,count=np.unique(label,return_counts=True)
    prob_y=count/len(label)
    entropy_y=-np.sum(prob_y*np.log2(prob_y))
    #entropy_y=-prob_y @ np.log2(prob_y)
    #在计算条件增益
    feature_col=feature[:,index]
    feat_value,feat_count=np.unique(feature_col,return_counts=True)
    cond_entropy=0
    for v,cnt in zip(feat_value,feat_count):
        sub_label=label[feature_col==v]
        sub_lab,sub_count=np.unique(sub_label,return_counts=True)
        sub_prob=sub_count/len(sub_label)
        sub_entropy=-np.sum(sub_prob*np.log2(sub_prob))
        cond_entropy+=(cnt/len(label))*sub_entropy
    # 3. 信息增益
    info_gain=entropy_y-cond_entropy
    return info_gain
```

```python
import numpy as np

from sklearn.tree import DecisionTreeClassifier
class RandomForestClassifier():
    def __init__(self, n_model=10):
        '''
        初始化函数
        '''
        #分类器的数量，默认为10
        self.n_model = n_model
        #用于保存模型的列表，训练好分类器后将对象append进去即可
        self.models = []
        #用于保存决策树训练时随机选取的列的索引
        self.col_indexs = []
    def fit(self, feature, label):
        '''
        训练模型
        :param feature: 训练集数据，类型为ndarray
        :param label: 训练集标签，类型为ndarray
        :return: None
        '''
        #************* Begin ************#
        n_samples, n_features = feature.shape
        # 特征子集大小（分类问题常用 sqrt）
        n_sub_features = int(np.log2(n_features))
        for _ in range(self.n_model):
            # 1. 对样本进行有放回抽样（Bootstrap）
            idx_samples = np.random.choice(n_samples, n_samples, replace=True)
            X_sample = feature[idx_samples]
            y_sample = label[idx_samples]
            
            # 2. 随机选择特征列（不重复）
            col_idx = np.random.choice(n_features, n_sub_features, replace=False)
            self.col_indexs.append(col_idx)
            
            # 3. 用选出的样本和特征训练一棵决策树
            tree = DecisionTreeClassifier()
            tree.fit(X_sample[:, col_idx], y_sample)
            self.models.append(tree)
        #************* End **************#
    def predict(self, feature):
        '''
        :param feature:测试集数据，类型为ndarray
        :return:预测结果，类型为ndarray，如np.array([0, 1, 2, 2, 1, 0])
        '''
        #************* Begin ************#
        preds = []
        for tree, col_idx in zip(self.models, self.col_indexs):
            pred = tree.predict(feature[:, col_idx])
            preds.append(pred)
        preds = np.array(preds)
        # 多数投票
        final_pred = []
        for i in range(feature.shape[0]):
            votes = preds[:, i]
            most_common = Counter(votes).most_common(1)[0][0]
            final_pred.append(most_common)
        return np.array(final_pred)
```

### k-means

```python
#encoding=utf8
import numpy as np
# 计算一个样本与数据集中所有样本的欧氏距离的平方
def euclidean_distance(one_sample, X):
    one_sample = one_sample.reshape(1, -1)
    distances = np.power(np.tile(one_sample, (X.shape[0], 1)) - X, 2).sum(axis=1)
    return distances#m,1型
class Kmeans():
    """Kmeans聚类算法.
    Parameters:
    -----------
    k: int
        聚类的数目.
    max_iterations: int
        最大迭代次数. 
    varepsilon: float
        判断是否收敛, 如果上一次的所有k个聚类中心与本次的所有k个聚类中心的差都小于varepsilon, 
        则说明算法已经收敛
    """
    def __init__(self, k=2, max_iterations=500, varepsilon=0.0001):
        self.k = k
        self.max_iterations = max_iterations
        self.varepsilon = varepsilon
        np.random.seed(1)
    #********* Begin *********#
    # 从所有样本中随机选取self.k样本作为初始的聚类中心
    def init_random_centroids(self, X):
        m=X.shape[0]
        indices=np.random.choice(m,self.k,replace=False)
        return X[indices]
    # 返回距离该样本最近的一个中心索引[0, self.k)
    def _closest_centroid(self, sample, centroids):
        dis=euclidean_distance(sample,centroids)
        return np.argmin(dis)
    # 将所有样本进行归类，归类规则就是将该样本归类到与其最近的中心
    def create_clusters(self, centroids, X):
        clusters=[[] for _ in range(self.k)]
        for i,sample in enumerate(X):
            label=self._closest_centroid(sample,centroids)
            clusters[label].append(i)
        return clusters

    # 对中心进行更新
    def update_centroids(self, clusters, X):
        new_centroids = np.zeros((self.k, X.shape[1]))
        for i,clusters_indices in enumerate(clusters):
            if len(clusters_indices)==0:
                new_centroids[i]=X[np.random.choice(X.shape[0])]
            else:
                new_centroids[i]=np.mean(X[clusters_indices],axis=0)
        return new_centroids
    # 将所有样本进行归类，其所在的类别的索引就是其类别标签
    def get_cluster_labels(self, clusters, X):
        labels=np.zeros(X.shape[0],dtype=int)
        for label,indices in enumerate(clusters):
            labels[indices]=label
        return labels

    # 对整个数据集X进行Kmeans聚类，返回其聚类的标签
    def predict(self, X):
        # 从所有样本中随机选取self.k样本作为初始的聚类中心
        centroids=self.init_random_centroids(X)
        # 迭代，直到算法收敛(上一次的聚类中心和这一次的聚类中心几乎重合)或者达到最大迭代次数
        for _ in range(self.max_iterations):
            # 将所有进行归类，归类规则就是将该样本归类到与其最近的中心
            clusters=self.create_clusters(centroids,X)
            # 计算新的聚类中心
            new_centroids=self.update_centroids(clusters,X)
            # 如果聚类中心几乎没有变化，说明算法已经收敛，退出迭代
            movement=np.sum((new_centroids-centroids)**2)
            if movement<self.varepsilon:
                break
            centroids=new_centroids
        return self.get_cluster_labels(clusters,X)
    #********* End *********#
```

### EM算法单次迭代过程

```python
import numpy as np
from scipy import stats

def em_single(init_values, observations):
    """
    模拟抛掷硬币实验并估计在一次迭代中，硬币A与硬币B正面朝上的概率
    :param init_values: 硬币A与硬币B正面朝上的概率的初始值，类型为list，如[0.2, 0.7]代表硬币A正面朝上的概率为0.2，硬币B正面朝上的概率为0.7。
    :param observations: 抛掷硬币的实验结果记录，类型为list，每个元素是一个批次的结果列表（0表示反面，1表示正面）。
    :return: 将估计出来的硬币A和硬币B正面朝上的概率组成list返回。如[0.4, 0.6]表示你认为硬币A正面朝上的概率为0.4，硬币B正面朝上的概率为0.6。
    """
    #********* Begin *********#
    # 从初始值中取出硬币A和B的正面概率
    Pa, Pb = init_values[0], init_values[1]
    # 初始化用于累积期望统计量的变量（A硬币和B硬币的正面/反面期望次数）
    Ta_p = Ta_n = 0   # 硬币A的正面期望次数、反面期望次数
    Tb_p = Tb_n = 0   # 硬币B的正面期望次数、反面期望次数
    # 遍历每一批次的实验数据
    for batch in observations:
        # 当前批次中正面的次数（硬币正面为1）
        positive = sum(batch)
        # 当前批次的总抛掷次数
        times = len(batch)
        # 当前批次中反面的次数
        negative = times - positive
        # 计算在当前参数（Pa, Pb）下，这批结果分别来自硬币A和硬币B的似然概率
        # 即 P(数据 | 硬币A) 和 P(数据 | 硬币B)
        lik_A = Pa ** positive * (1 - Pa) ** negative
        lik_B = Pb ** positive * (1 - Pb) ** negative
        # E步：计算这批数据属于硬币A的后验概率（归一化），属于B的概率为1减去它
        prob_A_batch = lik_A / (lik_A + lik_B)
        prob_B_batch = 1 - prob_A_batch
        # 根据后验概率，将本批次的正反面次数按期望分配到硬币A和B的计数上
        # 硬币A的期望正面次数 += prob_A_batch * positive
        Ta_p += prob_A_batch * positive
        # 硬币A的期望反面次数 += prob_A_batch * negative
        Ta_n += prob_A_batch * negative
        # 硬币B的期望正面次数 += prob_B_batch * positive
        Tb_p += prob_B_batch * positive
        # 硬币B的期望反面次数 += prob_B_batch * negative
        Tb_n += prob_B_batch * negative
    # M步：根据累积的期望次数，重新估计硬币A和B的正面概率
    new_Pa = Ta_p / (Ta_n + Ta_p)   # 硬币A的正面概率 = A的正面期望次数 / A的总期望次数
    new_Pb = Tb_p / (Tb_p + Tb_n)   # 硬币B的正面概率 = B的正面期望次数 / B的总期望次数
    # 返回新一轮估计的概率
    return [new_Pa, new_Pb]
    #********* End *********#
```

### PCA(数据没有标准化，只求了均值的版本)

- **情况 A（够用）**：如果所有特征都是同一种单位（比如像素值 0~255，或者都是厘米、千克），**只做中心化就够了**。因为 PCA 关注的是方差结构，无量纲问题时无需缩放。
- **情况 B（不够，需要标准化）**：如果特征量纲不同（比如一个特征是“年龄 20~40”，另一个是“工资 10000~50000”），**必须加上 `data = (data - u) / std`**。因为工资的方差天然比年龄大几个数量级，不做标准化的话，PCA 会被工资特征完全主导，忽略年龄的贡献。

```python
import numpy as np
def pca(data, k):
    '''
    对data进行PCA，并将结果返回
    :param data:数据集，类型为ndarray
    :param k:想要降成几维，类型为int
    :return: 降维后的数据，类型为ndarray
    '''
    #********* Begin *********#
    u=np.mean(data,axis=0)
    after_mean=data-u
    #cov函数的输入希望是行代表特征，列代表数据的矩阵，所以要转置
    #写法一
    cov=np.cov(after_mean.T)
    #写法二
    #cov=np.cov(after_mean,rowvar=False)
    #写法三
    #n_samples = after_mean.shape[0]
	#cov = (after_mean.T @ after_mean) / (n_samples - 1)
    value,vector=np.linalg.eig(cov)
    index=np.argsort(-value)
    selected_vectors=vector[:,index[:k]]
    result=after_mean.dot(selected_vectors)
    return result
    #********* End *********#
```
